\section{Methods}
\label{sec:methods}

We compare four approaches under the strict protocol of
Section~\ref{sec:preliminaries}.

\paragraph{ERM (baseline).}
A single model trained by SGD on one bootstrap of the canonical
training set.

\paragraph{Deep ensemble.}
$K$ models trained on the \emph{same} bootstrap with different
initialisation seeds; predictions are averaged at deployment.  This
captures parameter-distribution variance only.

\paragraph{$K$-bootstrap bagging.}
$K$ models trained on $K$ \emph{independent} bootstraps of the canonical
training set, predictions averaged at deployment.  This is the classic
\citet{TODO_breiman1996} estimator.  The inductive bias is purely
averaging---no auxiliary loss---but, critically, the data axis is
varied along with parameters.

\paragraph{Twin-indep with bootstrap-consistency loss.}
Two networks $\theta_A, \theta_B$ are trained jointly on independent
bootstraps $S_A, S_B$.  At each step we draw one mini-batch from each
loader, forward both networks on both mini-batches, and minimise
\begin{equation}
  \mathcal{L} \;=\;
    \mathcal{L}_{\mathrm{CE}}(\theta_A; S_A^{(t)}) \;+\;
    \mathcal{L}_{\mathrm{CE}}(\theta_B; S_B^{(t)}) \;+\;
    \lambda \cdot \mathcal{L}_{\mathrm{cons}},
\end{equation}
where $\mathcal{L}_{\mathrm{CE}}$ is standard cross-entropy on each
network's own bootstrap, and the consistency term
$\mathcal{L}_{\mathrm{cons}}$ is the symmetric KL between the two
networks' softmax distributions evaluated on the union of the two
mini-batches.  At deployment, predictions are averaged.

\paragraph{Algorithm.}
\begin{algorithm}[H]
  \caption{Twin-indep training (one epoch).}
  \label{alg:twin-indep}
  \begin{algorithmic}[1]
    \Require canonical training set $S$, networks $\theta_A, \theta_B$,
             optimisers $\mathrm{opt}_A, \mathrm{opt}_B$, weight $\lambda$
    \State Sample $S_A, S_B \sim \mathrm{Bootstrap}(S)$
           \Comment{$N$ draws with replacement, independent}
    \State Build dataloaders $L_A$ over $S_A$, $L_B$ over $S_B$
    \For{$(B_A, B_B)$ zipped from $(L_A, L_B)$}
      \State $\hat{p}_A^{(A)} \gets f_{\theta_A}(B_A);\quad \hat{p}_B^{(A)} \gets f_{\theta_B}(B_A)$
      \State $\hat{p}_A^{(B)} \gets f_{\theta_A}(B_B);\quad \hat{p}_B^{(B)} \gets f_{\theta_B}(B_B)$
      \State $\mathcal{L}_{\mathrm{CE}} \gets \mathrm{CE}(\hat{p}_A^{(A)}, y_A) + \mathrm{CE}(\hat{p}_B^{(B)}, y_B)$
      \State $\mathcal{L}_{\mathrm{cons}} \gets \tfrac{1}{2}\big[\mathrm{symKL}(\hat{p}_A^{(A)}, \hat{p}_B^{(A)}) + \mathrm{symKL}(\hat{p}_A^{(B)}, \hat{p}_B^{(B)})\big]$
      \State $\mathcal{L} \gets \mathcal{L}_{\mathrm{CE}} + \lambda \, \mathcal{L}_{\mathrm{cons}}$
      \State Backprop $\mathcal{L}$ through both networks; step $\mathrm{opt}_A, \mathrm{opt}_B$
    \EndFor
  \end{algorithmic}
\end{algorithm}

\paragraph{Mechanism.}
Without the consistency term ($\lambda{=}0$) the procedure reduces to
$K{=}2$ bagging: each network fits its own bootstrap and they are
averaged at deployment.  Adding the consistency term penalises
divergence on every example seen in either mini-batch, including the
$\sim 40\%$ of indices that appear in both bootstraps in
expectation.\footnote{Bootstrap-with-replacement at size $N$ covers
${\sim}63\%$ of unique indices.  Two independent bootstraps share
${\sim}0.63 \times 0.63 \approx 0.40$ in expectation.}  The objective
is therefore an explicit pull toward decision functions that are
invariant to the specific bootstrap.  We measure empirically
(Section~\ref{sec:experiments}) that this bias transfers to invariance
under \emph{fresh} bootstraps: cross-sample fragility on held-out
bootstrap pairs is reduced by a factor not captured by bagging alone.
We do not claim that bootstrap-invariance \emph{implies}
cross-sample-invariance in general; the connection is theoretical
motivation, not a proof.

\paragraph{Hyperparameter selection (no test-set tuning).}
A single development dataset (BACE) is used to select $\lambda$ via
the pre-registered rule: largest $\lambda$ in
$\{1, 3, 10, 30, 100, 300\}$ such that BACE id-accuracy is within
$0.02$ of ERM id-accuracy.  This rule yields $\lambda{=}300$.  The
chosen $\lambda$ is frozen and applied unchanged to every held-out
dataset.  $K$ for ensemble methods is set a priori to $5$ following
\citet{TODO_lakshminarayanan2017}; we additionally report $K{=}2$ for
compute-matched comparison against twin-indep.
